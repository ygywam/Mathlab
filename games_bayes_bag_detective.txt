<script>
(function(){
'use strict';
var APP_ID='bayes-bag-detective',APP_TITLE='베이즈 정리 주머니 추리',BOARD='board-bayes-bag-detective',TOTAL=6;
var state={mounted:false,host:null,diff:'normal',round:1,score:0,combo:0,finished:false,bags:[],color:'R',post:[],picked:null,hintUsed:false,revealed:false,pending:[]};
function byId(id){return document.getElementById(id)}
function setText(id,v){var el=byId(id);if(el)el.textContent=String(v)}
function esc(s){return String(s==null?'':s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')}
var toastT=null;function toast(m,t){var el=byId('bbd-toast');if(!el)return;el.textContent=m;el.className='bbd-toast on'+(t?' '+t:'');if(toastT)clearTimeout(toastT);toastT=setTimeout(function(){el.className='bbd-toast'},2200)}
function makeBags(){var n=state.diff==='easy'?2:3;var bags=[],i,pr=1/n;
for(i=0;i<n;i++){var red=2+Math.floor(Math.random()*6),blue=2+Math.floor(Math.random()*6);if(state.diff==='easy'&&i===0){red=7;blue=3}if(state.diff==='easy'&&i===1){red=2;blue=8}bags.push({name:String.fromCharCode(65+i),prior:pr,red:red,blue:blue})}return bags}
function likelihood(bag,color){var t=bag.red+bag.blue;return color==='R'?bag.red/t:bag.blue/t}
function posterior(bags,color){var raw=bags.map(function(b){return b.prior*likelihood(b,color)});var s=raw.reduce(function(a,b){return a+b},0)||1;return raw.map(function(v){return v/s})}
function startRound(){state.bags=makeBags();state.color=Math.random()<0.5?'R':'B';state.post=posterior(state.bags,state.color);state.picked=null;state.hintUsed=false;state.revealed=false;render()}
function bestIdx(){var bi=0,i;for(i=1;i<state.post.length;i++)if(state.post[i]>state.post[bi])bi=i;return bi}
function render(){setText('bbd-round',state.round+'/'+TOTAL);setText('bbd-score',state.score);setText('bbd-obs',state.color==='R'?'🔴 빨간 공':'🔵 파란 공');
var wrap=byId('bbd-bags');if(!wrap)return;
wrap.innerHTML=state.bags.map(function(b,i){var cls=state.picked===i?(state.revealed?(i===bestIdx()?' ok':' bad'):' on'):'';var postH=state.revealed||state.hintUsed?('<div style="margin-top:6px;font-size:.85rem">사후 '+(state.post[i]*100).toFixed(1)+'%</div><div class="bbd-barrow"><div class="bbd-barfill" style="width:'+(state.post[i]*100)+'%"></div></div>'):'<div style="margin-top:6px;color:#94a3b8;font-size:.85rem">사후 ?</div>';
return '<div class="bbd-bag'+cls+'" data-i="'+i+'"><div style="font-size:1.4rem">🎒</div><div style="font-weight:900">주머니 '+b.name+'</div><div style="font-size:.88rem;margin-top:4px">사전 '+(b.prior*100).toFixed(0)+'%</div><div style="font-size:.88rem">🔴'+b.red+' 🔵'+b.blue+'</div>'+postH+'</div>'}).join('');
document.querySelectorAll('.bbd-bag').forEach(function(el){el.onclick=function(){if(state.finished||state.revealed)return;state.picked=parseInt(el.getAttribute('data-i'),10);render()}});
document.querySelectorAll('.bbd-diff').forEach(function(b){b.classList.toggle('on',b.getAttribute('data-d')===state.diff)});
var formula=byId('bbd-formula');if(formula){if(state.revealed||state.hintUsed){var lines=state.bags.map(function(b,i){return 'P('+b.name+'|D)=('+likelihood(b,state.color).toFixed(3)+'×'+b.prior.toFixed(2)+')/Σ = '+state.post[i].toFixed(3)});formula.textContent=lines.join(' | ')}else formula.textContent='P(Bi|D)=P(D|Bi)P(Bi)/Σ P(D|Bj)P(Bj)  — 관측 후 제출하세요.'}
}
function submit(){if(state.finished||state.revealed)return;if(state.picked==null){toast('주머니를 선택하세요','err');return}state.revealed=true;var bi=bestIdx(),sc=0;if(state.picked===bi){sc+=30;state.combo++;toast('최대 사후 적중 +30','ok')}else{state.combo=0;toast('아쉽네요','err')}
var inp=byId('bbd-num');if(inp&&inp.value!==''){var g=parseFloat(inp.value);if(!isNaN(g)){if(g>1)g=g/100;var er=Math.abs(g-state.post[state.picked]);if(er<=0.02)sc+=50;else if(er<=0.05)sc+=25}}if(state.hintUsed)sc=Math.max(0,sc-10);if(state.combo>=2)sc+=(state.combo-1)*5;state.score+=sc;setText('bbd-score',state.score);setText('bbd-msg','정답 주머니 '+state.bags[bi].name+' (사후 '+(state.post[bi]*100).toFixed(1)+'%). +'+sc+'점. 사전 확률이 달라도 관측이 믿음을 갱신합니다.');render();var t=setTimeout(function(){if(state.round>=TOTAL)finish();else{state.round++;startRound()}},1400);state.pending.push(t)}
function finish(){state.finished=true;toast('종료 '+state.score+'점','ok');openRank(true)}
function openRank(reg){var m=byId('bbd-modal');var list=[];try{list=JSON.parse(localStorage.getItem('mathlab_lb_bayes')||'[]')}catch(e){}
var rows=list.slice(0,5).map(function(it,i){return '<div style="display:flex;justify-content:space-between;padding:6px 0;border-bottom:1px solid #e2e8f0"><span>'+(i+1)+' '+esc(it.nickname)+'</span><b style="color:#0284c7">'+it.score+'</b></div>'}).join('')||'<p style="color:#64748b">기록 없음</p>';
var regH=reg?'<p><b>최종 '+state.score+'점</b></p><input id="bbd-nick" maxlength="12" placeholder="닉네임" style="width:100%;padding:10px;border:1px solid #cbd5e1;border-radius:8px"><button class="bbd-btn p" id="bbd-save" style="width:100%;margin-top:8px">등록</button>':'';
m.innerHTML='<div class="box"><h3>🏆 명예의 전당</h3>'+regH+rows+'<button class="bbd-btn" id="bbd-close" style="width:100%;margin-top:10px">닫기</button></div>';m.className='bbd-modal on';byId('bbd-close').onclick=function(){m.className='bbd-modal'};
var sb=byId('bbd-save');if(sb)sb.onclick=function(){var nm=(byId('bbd-nick').value||'익명').trim().slice(0,12);try{if(typeof submitRanking==='function')submitRanking(APP_TITLE,state.score,true,nm,BOARD)}catch(e){}try{list.push({nickname:nm,score:state.score});list.sort(function(a,b){return b.score-a.score});localStorage.setItem('mathlab_lb_bayes',JSON.stringify(list.slice(0,20)))}catch(e){}toast('저장','ok');m.className='bbd-modal';loadRank()}}
function loadRank(){var el=byId(BOARD);if(!el)return;if(typeof loadGameLeaderboard==='function'){try{loadGameLeaderboard(APP_TITLE,true,BOARD);return}catch(e){}}var list=[];try{list=JSON.parse(localStorage.getItem('mathlab_lb_bayes')||'[]')}catch(e){}el.innerHTML=list.length?list.slice(0,5).map(function(it,i){return '<div style="display:flex;justify-content:space-between;padding:8px;background:#f8fafc;border:1px solid #e2e8f0;border-radius:8px;margin-bottom:6px"><span>'+(i+1)+' '+esc(it.nickname)+'</span><b style="color:#0284c7">'+it.score+'</b></div>'}).join(''):'<p style="text-align:center;color:#94a3b8">기록 없음</p>'}
function html(){return '<div class="game-bayes-bag-detective" id="bbd-root"><div class="bbd-h"><h2 class="bbd-t">🎒 베이즈 정리 주머니 추리</h2><div style="display:flex;gap:6px"><button type="button" class="bbd-btn" id="bbd-guide">📖</button><button type="button" class="bbd-btn" id="bbd-home">🏠 홈</button></div></div>'+
'<div class="bbd-bar"><button type="button" class="bbd-diff" data-d="easy">쉬움</button><button type="button" class="bbd-diff on" data-d="normal">보통</button><button type="button" class="bbd-diff" data-d="hard">도전</button></div>'+
'<div class="bbd-bar"><span class="bbd-chip">라운드 <b id="bbd-round">1/6</b></span><span class="bbd-chip">점수 <b id="bbd-score">0</b></span><span class="bbd-chip">관측 <b id="bbd-obs">—</b></span></div>'+
'<div class="bbd-box"><div style="font-weight:800;margin-bottom:8px">주머니 구성 (사전확률·공 개수)</div><div class="bbd-bags" id="bbd-bags"></div></div>'+
'<div class="bbd-box"><div class="bbd-msg" id="bbd-msg">공이 나왔습니다. 어느 주머니에서 뽑혔을 가능성이 가장 클까요?</div><div id="bbd-formula" style="font-family:monospace;font-size:.85rem;color:#0369a1;margin-top:8px"></div>'+
'<div style="margin-top:10px">선택 주머니 사후확률 입력(0~1, 선택): <input id="bbd-num" type="number" step="0.01" min="0" max="1" style="width:100px;padding:8px;border:1px solid #cbd5e1;border-radius:8px"></div>'+
'<div class="bbd-bar" style="margin-top:10px"><button type="button" class="bbd-btn" id="bbd-hint">💡 공식 힌트(-10)</button><button type="button" class="bbd-btn g" id="bbd-submit">✅ 제출</button><button type="button" class="bbd-btn" id="bbd-restart">🔄</button><button type="button" class="bbd-btn p" id="bbd-rank">🏆</button></div></div>'+
'<div class="bbd-note">기준율 무시 주의: 사전확률이 작아도 우도가 크면 사후가 커질 수 있고, 그 반대도 성립합니다.</div>'+
'<div class="bbd-box" style="margin-top:10px"><div style="font-weight:900;margin-bottom:8px">🏆 명예의 전당</div><div id="'+BOARD+'"></div></div>'+
'<div class="bbd-toast" id="bbd-toast"></div><div class="bbd-modal" id="bbd-modal"></div></div>'}
function bind(){byId('bbd-home').onclick=goHome;byId('bbd-submit').onclick=submit;byId('bbd-rank').onclick=function(){openRank(state.finished)};byId('bbd-restart').onclick=function(){state.finished=false;state.round=1;state.score=0;state.combo=0;startRound()};
byId('bbd-hint').onclick=function(){if(state.hintUsed||state.revealed)return;state.hintUsed=true;render();toast('공식·사후 공개(-10)','err')};

document.querySelectorAll('.bbd-diff').forEach(function(b){b.onclick=function(){state.diff=b.getAttribute('data-d');state.finished=false;state.round=1;state.score=0;state.combo=0;startRound()}})}
function mount(c){var host=null;if(typeof c==='string'){try{host=document.querySelector(c)}catch(e){}}else if(c instanceof HTMLElement)host=c;if(!host)host=document.getElementById('mathlab-view')||document.getElementById('new-root');
if(!host||host===document.body){host=document.getElementById('bayes-bag-detective-force-host');if(!host){host=document.createElement('div');host.id='bayes-bag-detective-force-host';document.body.appendChild(host)}}host.style.display='';host.style.minHeight='60vh';state.host=host;state.mounted=true;host.innerHTML=html();bind();startRound();loadRank()}
function destroy(){state.pending.forEach(clearTimeout);state.pending=[];state.mounted=false;if(state.host)try{state.host.innerHTML=''}catch(e){}state.host=null}
function goHome(){destroy();try{if(window.LegacyBridge&&window.LegacyBridge.backHome){window.LegacyBridge.backHome();return}}catch(e){}var fh=document.getElementById('bayes-bag-detective-force-host');if(fh){fh.style.display='none';fh.innerHTML=''}var mv=document.getElementById('mathlab-view');if(mv){mv.style.display='none';mv.innerHTML=''}var ag=document.querySelector('.app-grid'),ds=document.getElementById('dashboard-screen');if(ag)ag.style.display='';if(ds)ds.style.display='';if(typeof window.goHome==='function')try{window.goHome()}catch(e){}}
var API={mount:mount,resetGame:function(){state.finished=false;state.round=1;state.score=0;state.combo=0;startRound()},destroy:destroy,goHome:goHome};
function reg(){if(typeof window.MathLabGames==='undefined'||!window.MathLabGames.register)return;try{if(window.MathLabGames.get&&window.MathLabGames.get(APP_ID))return;window.MathLabGames.register({id:APP_ID,title:APP_TITLE,icon:'🎒',sourceType:'new',mount:function(c,x){API.mount(c,x)},reset:function(){API.resetGame()},destroy:function(){API.destroy()},goHome:function(){API.goHome()}})}catch(e){console.warn(e)}}
reg();window.addEventListener('DOMContentLoaded',reg);setTimeout(reg,100);setTimeout(reg,500);
window.BayesBagDetective=API;window.initBayesBagDetectiveGame=function(){reg();try{if(window.LegacyBridge){window.LegacyBridge.hideAllLegacySections();if(window.LegacyBridge.showNewRoot)window.LegacyBridge.showNewRoot()}}catch(e){}if(window.MathLabGames&&window.MathLabGames.mount){try{window.MathLabGames.mount(APP_ID);return}catch(e){}}API.mount(document.getElementById('mathlab-view'))};window.launchBayesBagDetectiveGame=window.initBayesBagDetectiveGame;
})();
